{{- $src := .Destination -}} {{- if or (hasPrefix $src "https://") (hasPrefix $src "http://") -}} {{- /* Copy remote images (Gofylo hotlinks its own bucket) into the build so the site serves them; a missing image fails the build */ -}} {{- with resources.GetRemote $src -}} {{- with .Err -}} {{- errorf "render-image: fetching %s failed: %s" $src . -}} {{- else -}} {{- if eq .MediaType.MainType "image" -}} {{- $src = .RelPermalink -}} {{- else -}} {{- errorf "render-image: %s returned %s, not an image" $src .MediaType.Type -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "render-image: %s was not found" $src -}} {{- end -}} {{- end -}} {{- /* Trims the trailing newline Hugo would print after every image. */ -}}
